def now_iso() -> str:
    return datetime.datetime.utcnow().replace(microsecond=0).isoformat() + "Z"

def money(cents: int) -> str:
    return f"${cents/100:,.2f}"

def slugify(s: str) -> str:
    s = s.lower().strip()
    s = re.sub(r"[^a-z0-9]+", "-", s).strip("-")
    return s

def get_user_from_cookie(req: Request) -> Optional[sqlite3.Row]:
    token = req.cookies.get("session")
    if not token:
        return None
    try:
        data = signer.loads(token)
        uid = int(data.get("uid"))
    except Exception:
        return None
    conn = db()
    user = conn.execute("SELECT * FROM users WHERE id = ?", (uid,)).fetchone()
    conn.close()
    return user

def require_user(req: Request) -> sqlite3.Row:
    u = get_user_from_cookie(req)
    if not u:
        raise HTTPException(status_code=401, detail="Please log in.")
    return u

def require_admin(req: Request) -> sqlite3.Row:
    u = require_user(req)
    if int(u["is_admin"]) != 1:
        raise HTTPException(status_code=403, detail="Admin only.")
    return u

def ensure_affiliate_code(user_id: int) -> str:
    conn = db()
    u = conn.execute("SELECT affiliate_code FROM users WHERE id = ?", (user_id,)).fetchone()
    if u and u["affiliate_code"]:
        conn.close()
        return u["affiliate_code"]
    # generate unique code
    for _ in range(20):
        code = "PB" + secrets.token_hex(4).upper()
        try:
            conn.execute("UPDATE users SET affiliate_code = ? WHERE id = ?", (code, user_id))
            conn.commit()
            conn.close()
            return code
        except sqlite3.IntegrityError:
            continue
    conn.close()
    raise RuntimeError("Could not generate affiliate code.")

def first_user_becomes_admin() -> bool:
    conn = db()
    count = conn.execute("SELECT COUNT(*) AS c FROM users").fetchone()["c"]
    conn.close()
    return count == 0

def set_session(resp: Response, user_id: int):
    token = signer.dumps({"uid": user_id, "ts": int(time.time())})
    resp.set_cookie("session", token, httponly=True, samesite="lax", secure=BASE_URL.startswith("https://"))

def clear_session(resp: Response):
    resp.delete_cookie("session")

                conn.execute(

def make_estimate_pdf_bytes(
    estimate_id: int,
    service: str,
    state: str,
    description: str,
    low_cents: int,
    high_cents: int,
    referral_url: str,
    watermark: bool
) -> bytes:
    import io
    buf = io.BytesIO()
    c = canvas.Canvas(buf, pagesize=letter)
    w, h = letter

    c.setFont("Helvetica-Bold", 18)
    c.drawString(50, h-60, "ProBid AI Estimate")
    c.setFont("Helvetica", 11)
    c.drawString(50, h-85, f"Estimate ID: {estimate_id}")
    c.drawString(50, h-105, f"Service: {service}   |   State: {state}")
    c.drawString(50, h-125, f"Range: {money(low_cents)}  –  {money(high_cents)}")

    c.setFont("Helvetica-Bold", 12)
    c.drawString(50, h-165, "Scope / Notes")
    c.setFont("Helvetica", 11)

    y = h-185
    for line in wrap_text(description, 92):
        c.drawString(50, y, line)
        y -= 14
        if y < 120:
            c.showPage()
            y = h-60

    # Viral footer
    c.setFont("Helvetica-Bold", 11)
    c.drawString(50, 90, "Generated by ProBid AI • Get 3 free estimates:")
    c.setFont("Helvetica", 10)
    c.drawString(50, 74, referral_url)

    # Optional QR
    if qrcode:
        try:
            img = qrcode.make(referral_url)
            import io
            img_buf = io.BytesIO()
            img.save(img_buf, format="PNG")
            img_buf.seek(0)
            from reportlab.lib.utils import ImageReader
            c.drawImage(ImageReader(img_buf), w-150, 55, width=85, height=85, mask='auto')
        except Exception:
            pass

    # Watermark for free plan
    if watermark:
        c.saveState()
        c.setFont("Helvetica-Bold", 48)
        c.setFillGray(0.85, 0.18)
        c.translate(70, 240)
        c.rotate(35)
        c.drawString(0, 0, "PROBID AI FREE")
        c.restoreState()

    c.showPage()
    c.save()
    return buf.getvalue()

def wrap_text(text: str, width: int) -> List[str]:
    words = text.split()
    lines, cur = [], []
    count = 0
    for w in words:
        if count + len(w) + (1 if cur else 0) > width:
            lines.append(" ".join(cur))
            cur = [w]
            count = len(w)
        else:
            cur.append(w)
            count += len(w) + (1 if cur else 0)
    if cur:
        lines.append(" ".join(cur))
    return lines

@app.post("/signup")
def signup(req: Request, email: str = Form(...), password: str = Form(...), ref: str = Form("")):
    email = email.strip().lower()
    if len(password) < 6:
        raise HTTPException(400, "Password too short.")
    conn = db()
    is_admin = 1 if first_user_becomes_admin() else 0
    ph = pwd.hash(password)
    try:
        conn.execute(
            "INSERT INTO users (email,password_hash,created_at,is_admin,commission_rate,referred_by_code) VALUES (?,?,?,?,?,?)",
            (email, ph, now_iso(), is_admin, DEFAULT_COMMISSION, (ref.strip() or None))
        )
        conn.commit()
    except sqlite3.IntegrityError:
        conn.close()
        raise HTTPException(400, "Email already exists.")
    user = conn.execute("SELECT * FROM users WHERE email = ?", (email,)).fetchone()

    # If referral code exists, attribute referral
    if user and user["referred_by_code"]:
        ref_code = user["referred_by_code"]
        referrer = conn.execute("SELECT * FROM users WHERE affiliate_code = ?", (ref_code,)).fetchone()
        if referrer:
            conn.execute("UPDATE users SET referred_by_user_id = ? WHERE id = ?", (referrer["id"], user["id"]))
            try:
                conn.execute(
                    "INSERT INTO referrals (referrer_user_id,referred_user_id,status,created_at) VALUES (?,?,?,?)",
                    (referrer["id"], user["id"], "signed_up", now_iso())
                )
            except sqlite3.IntegrityError:
                pass
            conn.commit()
    conn.close()

    resp = RedirectResponse("/app", status_code=302)
    set_session(resp, user["id"])
    return resp

@app.post("/login")
def login(email: str = Form(...), password: str = Form(...)):
    email = email.strip().lower()
    conn = db()
    user = conn.execute("SELECT * FROM users WHERE email = ?", (email,)).fetchone()
    conn.close()
    if not user or not pwd.verify(password, user["password_hash"]):
        raise HTTPException(401, "Invalid credentials.")
    resp = RedirectResponse("/app", status_code=302)
    set_session(resp, user["id"])
    return resp

def simple_pricing_engine(service: str, state: str, description: str) -> (int,int):
    """
    Placeholder pricing engine.
    Replace later with your AI + regional pricing model.
    """
    base = 120000  # $1,200.00
    s = service.lower()
    if "retaining" in s: base = 350000
    if "chimney" in s: base = 420000
    if "tuckpoint" in s: base = 180000
    if "concrete" in s: base = 260000
    if "kitchen" in s: base = 800000
    if "bath" in s: base = 650000
    # State multiplier (very rough)
    st = state.lower()
    mult = 1.0
    if st in ("california","new york"): mult = 1.35
    if st in ("texas","florida"): mult = 1.15
    if st in ("iowa","wisconsin","illinois","missouri","indiana","michigan","ohio","minnesota"): mult = 1.0
    base = int(base * mult)

    # Description length nudges size complexity
    complexity = min(len(description) / 400, 1.0)
    low = int(base * (0.85 + 0.10*complexity))
    high = int(base * (1.20 + 0.25*complexity))
    return low, high

@app.post("/stripe/webhook")
async def stripe_webhook(req: Request):
    if not stripe_ready():
        raise HTTPException(400, "Stripe not configured")
    stripe_init()

    payload = await req.body()
    sig = req.headers.get("stripe-signature", "")
    try:
        event = stripe.Webhook.construct_event(payload, sig, STRIPE_WEBHOOK_SECRET)
    except Exception as e:
        raise HTTPException(400, f"Webhook error: {str(e)}")

    etype = event["type"]
    obj = event["data"]["object"]

    # Handle subscription created/updated
    if etype in ("checkout.session.completed",):
        meta = obj.get("metadata") or {}
        user_id = int(meta.get("user_id", "0") or "0")
        plan = meta.get("plan")
        if user_id and plan in PLANS:
            conn = db()
            # Save customer/subscription IDs if present
            cust = obj.get("customer")
            sub = obj.get("subscription")
            conn.execute(
                "UPDATE users SET plan=?, stripe_customer_id=?, stripe_subscription_id=? WHERE id=?",
                (plan, cust, sub, user_id)
            )
            conn.commit()
            conn.close()

            # Mark referral as subscribed if user was referred
            conn = db()
            u = conn.execute("SELECT * FROM users WHERE id=?", (user_id,)).fetchone()
            if u and u["referred_by_user_id"]:
                conn.execute(
                    "UPDATE referrals SET status='subscribed' WHERE referrer_user_id=? AND referred_user_id=?",
                    (u["referred_by_user_id"], user_id)
                )
                conn.commit()
            conn.close()

    # Handle invoice paid → create affiliate earnings (recurring commissions)
    if etype in ("invoice.paid",):
        invoice_id = obj.get("id")
        customer_id = obj.get("customer")
        amount_paid = int(obj.get("amount_paid") or 0)
        currency = (obj.get("currency") or "usd").lower()

        conn = db()
        # Find user by customer id
        u = conn.execute("SELECT * FROM users WHERE stripe_customer_id=?", (customer_id,)).fetchone()
        if u and u["referred_by_user_id"]:
            affiliate_id = int(u["referred_by_user_id"])
            aff = conn.execute("SELECT * FROM users WHERE id=?", (affiliate_id,)).fetchone()
            rate = float(aff["commission_rate"]) if aff else DEFAULT_COMMISSION
            commission = int(round(amount_paid * rate))

            # Prevent duplicates for same invoice
            exists = conn.execute(
                "SELECT 1 FROM affiliate_earnings WHERE stripe_invoice_id=? AND affiliate_user_id=?",
                (invoice_id, affiliate_id)
            ).fetchone()

            if not exists and commission > 0:
                conn.execute(
                    "INSERT INTO affiliate_earnings (affiliate_user_id,referred_user_id,stripe_invoice_id,amount_cents,currency,status,created_at) VALUES (?,?,?,?,?,?,?)",
                    (affiliate_id, int(u["id"]), invoice_id, commission, currency, "pending", now_iso())
                )
        conn.commit()
        conn.close()

    return PlainTextResponse("ok", status_code=200)

@app.get("/admin/payouts.csv")
def admin_payouts_csv(req: Request):
    require_admin(req)
    conn = db()
    rows = conn.execute("""
      SELECT u.email AS affiliate_email,
             u.payout_method,
             u.payout_handle,
             SUM(e.amount_cents) AS total_cents
      FROM affiliate_earnings e
      JOIN users u ON u.id=e.affiliate_user_id
      WHERE e.status='pending'
      GROUP BY e.affiliate_user_id
      HAVING total_cents > 0
      ORDER BY total_cents DESC
    """).fetchall()
    conn.close()

    lines = ["affiliate_email,payout_method,payout_handle,total_usd"]
    for r in rows:
        lines.append(f"{r['affiliate_email']},{r['payout_method']},{r['payout_handle'] or ''},{r['total_cents']/100:.2f}")
    csv = "\n".join(lines)
    return Response(content=csv, media_type="text/csv", headers={"Content-Disposition":"attachment; filename=pending_payouts.csv"})
