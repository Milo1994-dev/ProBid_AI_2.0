  FOREIGN KEY(user_id) REFERENCES users(id)
);

CREATE INDEX IF NOT EXISTS idx_usage_day ON usage(day_key);
`);

// --- Helpers ---
function mustEnv(k: string) {
  const v = process.env[k];
  if (!v) throw new Error(`Missing required env var: ${k}`);
  return v;
}

function now() {
  return Date.now();
}

function dayKey(ts = Date.now()) {
  const d = new Date(ts);
  const yyyy = d.getUTCFullYear();
  const mm = String(d.getUTCMonth() + 1).padStart(2, "0");
  const dd = String(d.getUTCDate()).padStart(2, "0");
  return `${yyyy}-${mm}-${dd}`;
}

function requireAuth(req: express.Request, res: express.Response, next: express.NextFunction) {
  if (!req.session?.uid) return res.redirect("/login");
  next();
}

function getUser(uid: string) {
  return db.prepare("SELECT * FROM users WHERE id = ?").get(uid) as any | undefined;
}

function getSub(uid: string) {
  return db.prepare("SELECT * FROM subscriptions WHERE user_id = ?").get(uid) as any | undefined;
}

function isPaidActive(sub: any | undefined) {
  if (!sub) return false;
  return sub.status === "active" || sub.status === "trialing";
}

function getDailyUsage(uid: string) {
  const dk = dayKey();
  const row = db
    .prepare("SELECT estimates_count FROM usage WHERE user_id = ? AND day_key = ?")
    .get(uid, dk) as any | undefined;
  return row?.estimates_count ?? 0;
}

function incrementUsage(uid: string) {
  const dk = dayKey();
  const existing = db
    .prepare("SELECT estimates_count FROM usage WHERE user_id = ? AND day_key = ?")
    .get(uid, dk) as any | undefined;

  if (!existing) {
    db.prepare("INSERT INTO usage (user_id, day_key, estimates_count, updated_at) VALUES (?, ?, ?, ?)")
      .run(uid, dk, 1, now());
    return 1;
  } else {
    const next = (existing.estimates_count ?? 0) + 1;
    db.prepare("UPDATE usage SET estimates_count = ?, updated_at = ? WHERE user_id = ? AND day_key = ?")
      .run(next, now(), uid, dk);
    return next;
  }
}

// --- Paywall rules ---
const FREE_ESTIMATES_PER_DAY = 2; // free preview limit (tune)
function enforcePaywall(uid: string) {
  const sub = getSub(uid);
  if (isPaidActive(sub)) return { ok: true, tier: sub.price_id || "paid" };

  const used = getDailyUsage(uid);
  if (used >= FREE_ESTIMATES_PER_DAY) {
    return { ok: false, reason: "limit_reached", used };
  }
  return { ok: true, tier: "free", used };
}

app.post("/login", (req, res) => {
  const email = String(req.body.email || "").trim().toLowerCase();
  if (!email || !email.includes("@")) return res.status(400).send("Invalid email");

  // upsert user
  let user = db.prepare("SELECT * FROM users WHERE email = ?").get(email) as any | undefined;
  if (!user) {
    const uid = crypto.randomUUID();
    db.prepare("INSERT INTO users (id, email, created_at) VALUES (?, ?, ?)").run(uid, email, now());
    user = { id: uid, email };
  }

  req.session.uid = user.id;
  req.session.email = user.email;
  return res.redirect("/app");
});

// --- Stripe webhook handler ---
async function stripeWebhookHandler(req: express.Request, res: express.Response) {
  let event: Stripe.Event;

  try {
    const sig = req.headers["stripe-signature"] as string;
    event = stripe.webhooks.constructEvent(req.body, sig, STRIPE_WEBHOOK_SECRET);
  } catch (err: any) {
    console.error("Webhook signature verification failed:", err?.message || err);
    return res.status(400).send(`Webhook Error`);
  }

  try {
    switch (event.type) {
      case "checkout.session.completed": {
        const session = event.data.object as Stripe.Checkout.Session;
        const uid = String((session.metadata as any)?.user_id || "");
        const customer = String(session.customer || "");
        if (uid && customer) upsertStripeCustomer(uid, customer);
        break;
      }

      case "customer.subscription.created":
      case "customer.subscription.updated":
      case "customer.subscription.deleted": {
        const sub = event.data.object as Stripe.Subscription;
        const customerId = String(sub.customer);

        // map customer -> user_id
        const row = db
          .prepare("SELECT user_id FROM stripe_customers WHERE stripe_customer_id = ?")
          .get(customerId) as any | undefined;

        if (row?.user_id) {
          const userId = row.user_id as string;
          const status = sub.status;
          const priceId = sub.items.data?.[0]?.price?.id || null;
          const periodEnd = sub.current_period_end ? sub.current_period_end * 1000 : null;

          db.prepare(`
            INSERT INTO subscriptions (user_id, status, price_id, current_period_end, updated_at)
            VALUES (@user_id, @status, @price_id, @current_period_end, @updated_at)
            ON CONFLICT(user_id) DO UPDATE SET
              status=excluded.status,
              price_id=excluded.price_id,
              current_period_end=excluded.current_period_end,
              updated_at=excluded.updated_at
          `).run({
            user_id: userId,
            status,
            price_id: priceId,
            current_period_end: periodEnd,
            updated_at: now()
          });
        }
        break;
      }

      default:
        // ignore
        break;
    }

    res.json({ received: true });
  } catch (e) {
    console.error("Webhook handling error:", e);
    res.status(500).send("Webhook handler failed");
  }
}

function escapeHtml(s: string) {
  return s.replace(/[&<>"']/g, (c) => ({ "&": "&amp;", "<": "&lt;", ">": "&gt;", '"': "&quot;", "'": "&#39;" }[c] as string));
}

// --- Estimate engine (real, deterministic MVP) ---
// This is NOT “AI” yet, but it produces usable real-world estimates without external APIs.
// You can later swap in AI behind the same function with zero paywall changes.
function generateEstimate(jobType: string, market: string, details: string) {
  const marketMult: Record<string, number> = {
    midwest: 1.0,
    south: 0.95,
    west: 1.2,
    northeast: 1.25
  };

  const m = marketMult[market] ?? 1.0;

  // Base assumptions (tune these to match your real margins)
  const laborRate = 85 * m;   // $/hr blended crew
  const overheadMult = 1.22;  // overhead + small tools
  const profitMult = 1.18;    // profit target

  // Try to extract rough size numbers from text
  const nums = Array.from(details.matchAll(/(\d+(\.\d+)?)/g)).map(x => Number(x[1])).filter(n => Number.isFinite(n));
  const sizeHint = nums.length ? Math.max(...nums) : 0;

  let laborHours = 10;
  let material = 350 * m;

  switch (jobType) {
    case "tuckpointing":
      laborHours = clamp(12 + sizeHint * 0.5, 10, 80);
      material = 250 * m + sizeHint * 6 * m;
      break;

    case "chimney_rebuild":
      laborHours = 28 + (sizeHint > 10 ? 12 : 0);
      material = 900 * m + (sizeHint > 0 ? sizeHint * 15 * m : 0);
      break;

    case "retaining_wall":
      laborHours = clamp(20 + sizeHint * 0.8, 18, 120);
      material = 1100 * m + sizeHint * 18 * m;
      break;

    case "concrete_flatwork":
      laborHours = clamp(14 + sizeHint * 0.25, 12, 90);
      material = 800 * m + sizeHint * 8 * m;
      break;

    case "roof_repair":
      laborHours = clamp(10 + sizeHint * 0.35, 8, 60);
      material = 450 * m + sizeHint * 10 * m;
      break;

    default:
      laborHours = clamp(12 + sizeHint * 0.3, 10, 90);
      material = 500 * m + sizeHint * 7 * m;
      break;
  }

  const labor = laborHours * laborRate;
  const subtotal = (labor + material) * overheadMult;
  const total = subtotal * profitMult;

  const low = total * 0.9;
  const high = total * 1.15;

  return [
    `ESTIMATE (MVP Engine)`,
    `Job Type: ${jobType}`,
    `Market: ${market} (multiplier ${m.toFixed(2)})`,
    ``,
    `Assumptions:`,
    `- Blended labor rate: $${laborRate.toFixed(0)}/hr`,
    `- Estimated labor hours: ${laborHours.toFixed(1)} hrs`,
    `- Materials & disposals: $${material.toFixed(0)}`,
    `- Overhead multiplier: ${overheadMult.toFixed(2)}`,
    `- Profit multiplier: ${profitMult.toFixed(2)}`,
    ``,
    `Breakdown:`,
    `- Labor: $${labor.toFixed(0)}`,
    `- Materials: $${material.toFixed(0)}`,
    `- Subtotal (w/ overhead): $${subtotal.toFixed(0)}`,
    ``,
    `Recommended Bid Range: $${low.toFixed(0)} – $${high.toFixed(0)}`,
    ``,
    `Notes:`,
    `- This is a deterministic MVP estimator (fast + consistent).`,
    `- Replace with AI-based estimator later without changing paywall or billing.`,
  ].join("\n");
}

function clamp(n: number, min: number, max: number) {
  return Math.max(min, Math.min(max, n));
}

// --- Small override: ensure checkout/billing can create customer if missing ---
app.get("/checkout", requireAuth, async (req, res) => {
  const uid = req.session.uid!;
  const plan = String(req.query.plan || "pro");
  const priceId = plan === "business" ? PRICE_BIZ : PRICE_PRO;

  const email = req.session.email!;
  let customerId = getStripeCustomerIdOrNull(uid);
  if (!customerId) {
    customerId = await createStripeCustomer(email);
    upsertStripeCustomer(uid, customerId);
  }

  const session = await stripe.checkout.sessions.create({
    mode: "subscription",
    customer: customerId,
    line_items: [{ price: priceId, quantity: 1 }],
    allow_promotion_codes: true,
    success_url: `${APP_URL}/success`,
    cancel_url: `${APP_URL}/pricing`,
    metadata: { user_id: uid }
  });

  res.redirect(303, session.url!);
});

app.get("/billing", requireAuth, async (req, res) => {
  const uid = req.session.uid!;
  const email = req.session.email!;
  let customerId = getStripeCustomerIdOrNull(uid);
  if (!customerId) {
    customerId = await createStripeCustomer(email);
    upsertStripeCustomer(uid, customerId);
  }

  const portal = await stripe.billingPortal.sessions.create({
    customer: customerId,
    return_url: `${APP_URL}/app`
  });

  res.redirect(303, portal.url);
});
