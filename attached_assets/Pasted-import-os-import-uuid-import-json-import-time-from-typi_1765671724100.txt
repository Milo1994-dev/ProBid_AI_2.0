# -----------------------------
# VERY SIMPLE IN-MEMORY STATE (MVP)
# NOTE: resets if app restarts. Use DB in production.
# -----------------------------
VISITOR = {}  # visitor_id -> {"free_used": bool, "paid": bool, "paid_plan": str, "paid_at": int}

def get_or_set_visitor_id(req: Request) -> str:
    vid = req.cookies.get("visitor_id")
    if not vid:
        vid = str(uuid.uuid4())
    if vid not in VISITOR:
        VISITOR[vid] = {"free_used": False, "paid": False, "paid_plan": None, "paid_at": None}
    return vid

def is_paid(vid: str) -> bool:
    return bool(VISITOR.get(vid, {}).get("paid"))

def free_used(vid: str) -> bool:
    return bool(VISITOR.get(vid, {}).get("free_used"))

def mark_free_used(vid: str):
    VISITOR[vid]["free_used"] = True

def mark_paid(vid: str, plan: str):
    VISITOR[vid]["paid"] = True
    VISITOR[vid]["paid_plan"] = plan
    VISITOR[vid]["paid_at"] = int(time.time())

def require_base_url() -> str:
    # BASE_URL is strongly recommended for Stripe redirects.
    # If not set, we attempt to infer from request in create-checkout route, but setting is best.
    return BASE_URL.strip().rstrip("/")

# -----------------------------
# ESTIMATE ENGINE (MVP)
# Replace this with your real estimator / OpenAI call later
# -----------------------------
def generate_estimate_text(description: str) -> str:
    # MVP heuristic estimate template (fast + predictable)
    # You can wire OpenAI into here when you want.
    desc = description.strip()
    base = [
        "ESTIMATE SUMMARY (Quick-Use)",
        f"- Job Description: {desc[:280]}{'...' if len(desc) > 280 else ''}",
        "",
        "SCOPE (assumed):",
        "- Site visit, setup, basic protection",
        "- Labor + materials based on typical local conditions",
        "- Cleanup and haul-away (if applicable)",
        "",
        "LINE ITEMS (rough):",
        "- Labor: $650 – $1,850",
        "- Materials: $250 – $1,500",
        "- Overhead/Profit: 15% – 25%",
        "",
        "TOTAL RANGE (rough):",
        "$1,250 – $4,900",
        "",
        "NOTES:",
        "- This is a fast estimate. Final price depends on access, quantities, existing conditions, and schedule.",
        "- Add photos + measurements for a tighter number.",
    ]
    return "\n".join(base)

@app.post("/create-checkout-session")
async def create_checkout_session(request: Request, plan: str = Form(...)):
    vid = get_or_set_visitor_id(request)

    base = require_base_url()
    if not base:
        # Fallback: infer from request headers (not always reliable behind proxies)
        proto = request.headers.get("x-forwarded-proto", "https")
        host = request.headers.get("host", "")
        if host:
            base = f"{proto}://{host}"
        else:
            return JSONResponse({"error": "BASE_URL not set and could not infer host. Set BASE_URL in secrets."}, status_code=400)

    if not STRIPE_SECRET_KEY:
        return JSONResponse({"error": "Missing STRIPE_SECRET_KEY in env."}, status_code=400)

    if plan == "pro":
        if not STRIPE_PRICE_PRO:
            return JSONResponse({"error": "Missing STRIPE_PRICE_PRO in env."}, status_code=400)
        price_id = STRIPE_PRICE_PRO
    elif plan == "business":
        if not STRIPE_PRICE_BUSINESS:
            return JSONResponse({"error": "Missing STRIPE_PRICE_BUSINESS in env."}, status_code=400)
        price_id = STRIPE_PRICE_BUSINESS
    else:
        return JSONResponse({"error": "Invalid plan."}, status_code=400)

    # Stripe Checkout for subscription
    session = stripe.checkout.Session.create(
        mode="subscription",
        line_items=[{"price": price_id, "quantity": 1}],
        success_url=f"{base}/success?session_id={{CHECKOUT_SESSION_ID}}",
        cancel_url=f"{base}/pay?canceled=1",
        metadata={"visitor_id": vid, "plan": plan},
    )

    return RedirectResponse(session.url, status_code=303)

@app.get("/success", response_class=HTMLResponse)
async def success(request: Request, session_id: str):
    vid = get_or_set_visitor_id(request)

    if not STRIPE_SECRET_KEY:
        return JSONResponse({"error": "Missing STRIPE_SECRET_KEY in env."}, status_code=400)

    # Verify the Stripe session
    try:
        session = stripe.checkout.Session.retrieve(session_id)
        # session.payment_status may not exist for subscriptions, but session.status + subscription indicates success
        meta = session.get("metadata", {}) or {}
        plan = meta.get("plan", "pro")
        visitor_id_from_meta = meta.get("visitor_id")

        # Only unlock if this session belongs to the same visitor OR if metadata is missing (MVP tolerance)
        if visitor_id_from_meta and visitor_id_from_meta != vid:
            # If cookie changed, still allow unlocking the visitor_id in metadata:
            vid = visitor_id_from_meta
            if vid not in VISITOR:
                VISITOR[vid] = {"free_used": True, "paid": False, "paid_plan": None, "paid_at": None}

        mark_paid(vid, plan)

    except Exception as e:
        return JSONResponse({"error": f"Could not verify Stripe session: {str(e)}"}, status_code=400)

    body = f"""
    <div class="card">
      <h1>Unlocked ✅</h1>
      <p class="muted">You now have unlimited estimates. You can go back and keep generating bids.</p>
      <div class="row">
        <a href="/"><button>Go to Estimates</button></a>
        <a href="/pay"><button class="secondary" type="button">View Plan</button></a>
      </div>
    </div>
    """
    html = page_template(f"{APP_NAME} – Success", body)
    resp = HTMLResponse(html)
    resp.set_cookie("visitor_id", vid, httponly=True, samesite="lax", max_age=60*60*24*365)
    return resp
