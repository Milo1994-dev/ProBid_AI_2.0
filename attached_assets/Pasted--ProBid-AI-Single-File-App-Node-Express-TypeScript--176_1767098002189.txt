function escapeHtml(s: string) {
  return (s ?? "")
    .replaceAll("&", "&amp;")
    .replaceAll("<", "&lt;")
    .replaceAll(">", "&gt;")
    .replaceAll('"', "&quot;")
    .replaceAll("'", "&#039;");
}

function getOrCreateUser(req: express.Request, res: express.Response) {
  let uid = req.cookies["pb_uid"];
  if (!uid || typeof uid !== "string") {
    uid = uuidv4();
    res.cookie("pb_uid", uid, { httpOnly: true, sameSite: "lax" });
  }

  let user = qGetUser.get(uid);
  if (!user) {
    const referral_code = randomReferralCode();
    qInsertUser.run({
      id: uid,
      created_at: nowISO(),
      email: null,
      paid: 0,
      free_used: 0,
      referral_code,
      referral_credits: 0,
      referred_by: null,
    });
    user = qGetUser.get(uid);
    logEvent(uid, "user_created", { referral_code });
  }
  return user as any;
}

// Referral capture via query param ?ref=CODE
function captureReferral(req: express.Request, user: any) {
  const ref = (req.query.ref as string | undefined)?.trim();
  if (!ref) return;

  // only set once
  if (user.referred_by && user.referred_by !== "") return;

  const refOwner = db.prepare(`SELECT * FROM users WHERE referral_code = ?`).get(ref);
  if (refOwner && refOwner.id !== user.id) {
    qSetReferredBy.run(ref, user.id);
    // Give 1 credit to new user, and 1 to referrer
    qIncReferralCredit.run(user.id);
    qIncReferralCredit.run(refOwner.id);
    logEvent(user.id, "referral_captured", { ref });
    logEvent(refOwner.id, "referral_earned", { from_user: user.id });
  }
}

function canCreateEstimate(user: any) {
  const freeLimit = 3;
  if (user.paid === 1) return { ok: true, reason: "paid" };

  // referral credits can be used first
  if (Number(user.referral_credits || 0) > 0) return { ok: true, reason: "referral_credit" };

  const used = Number(user.free_used || 0);
  if (used < freeLimit) return { ok: true, reason: "free" };
  return { ok: false, reason: "paywall" };
}

const mailer =
  SMTP_HOST && SMTP_USER && SMTP_PASS
    ? nodemailer.createTransport({
        host: SMTP_HOST,
        port: SMTP_PORT,
        secure: SMTP_PORT === 465,
        auth: { user: SMTP_USER, pass: SMTP_PASS },
      })
    : null;

async function sendEmail(to: string, subject: string, html: string) {
  if (!mailer) return; // silently skip if not configured
  await mailer.sendMail({ from: SMTP_FROM, to, subject, html });
}

app.post("/estimate", async (req, res) => {
  const user = getOrCreateUser(req, res);

  // refresh user record
  const fresh = qGetUser.get(user.id) as any;

  const eligibility = canCreateEstimate(fresh);
  if (!eligibility.ok) {
    logEvent(fresh.id, "paywall_blocked", { reason: eligibility.reason });
    res.redirect("/paywall");
    return;
  }

  const trade = String(req.body.trade || "general").trim();
  const scope = String(req.body.scope || "").trim().slice(0, 2000);

  const labor_hours = Number(req.body.labor_hours || 0);
  const labor_rate = Number(req.body.labor_rate || 0);
  const materials = Number(req.body.materials || 0);
  const overhead_pct = Number(req.body.overhead_pct || 0);
  const profit_pct = Number(req.body.profit_pct || 0);

  const client_name = String(req.body.client_name || "").trim().slice(0, 140);
  const client_email = String(req.body.client_email || "").trim().slice(0, 140);

  // Very simple pricing math (replace with your AI estimator later)
  const labor = labor_hours * labor_rate;
  const subtotal = labor + materials;
  const overhead = subtotal * (overhead_pct / 100);
  const profit = subtotal * (profit_pct / 100);
  const total = Math.max(0, subtotal + overhead + profit);

  const id = uuidv4();
  qInsertEstimate.run({
    id,
    user_id: fresh.id,
    created_at: nowISO(),
    trade,
    scope: scope || "Scope not provided.",
    labor_hours: isFinite(labor_hours) ? labor_hours : 0,
    labor_rate: isFinite(labor_rate) ? labor_rate : 0,
    materials: isFinite(materials) ? materials : 0,
    overhead_pct: isFinite(overhead_pct) ? overhead_pct : 0,
    profit_pct: isFinite(profit_pct) ? profit_pct : 0,
    total: isFinite(total) ? total : 0,
    client_name: client_name || null,
    client_email: client_email || null,
  });

  // consume credit
  if (eligibility.reason === "referral_credit") {
    qDecReferralCredit.run(fresh.id);
  } else if (eligibility.reason === "free") {
    qIncFreeUsed.run(fresh.id);
  }

  logEvent(fresh.id, "estimate_created", {
    trade, total, credit_used: eligibility.reason,
  });

  // Optional email "Estimate ready"
  const latestUser = qGetUser.get(fresh.id) as any;
  const toEmail = client_email || latestUser.email;
  if (toEmail && toEmail.includes("@")) {
    try {
      await sendEmail(
        toEmail,
        "Your ProBid AI estimate is ready",
        `<div style="font-family:system-ui">
          <h2>Your estimate is ready</h2>
          <p>Total: <b>$${Number(total).toFixed(2)}</b></p>
          <p><a href="${BASE_URL}/pdf/${id}">Download the proposal PDF</a></p>
          <p style="opacity:.75">Generated by ProBid AI • ${BASE_URL}</p>
        </div>`
      );
      logEvent(fresh.id, "email_sent_estimate_ready", { toEmail });
    } catch (e: any) {
      logEvent(fresh.id, "email_failed", { msg: String(e?.message || e) });
    }
  }

  res.redirect(`/success/${id}`);
});

    id: uuidv4(),
    created_at: nowISO(),
    user_id: user.id,
    checkout_session_id: session.id,
    status: "created",
  });

  res.redirect(303, session.url!);
});

// Optional Stripe webhook (recommended)
app.post("/stripe/webhook", bodyParser.raw({ type: "application/json" }), (req, res) => {
  if (!stripe || !STRIPE_WEBHOOK_SECRET) {
    return res.status(400).send("Webhook not configured");
  }
  const sig = req.headers["stripe-signature"] as string;

  let event: Stripe.Event;
  try {
    event = stripe.webhooks.constructEvent(req.body, sig, STRIPE_WEBHOOK_SECRET);
  } catch (err: any) {
    return res.status(400).send(`Webhook Error: ${err.message}`);
  }

  if (event.type === "checkout.session.completed") {
    const session = event.data.object as Stripe.Checkout.Session;
    const userId = (session.metadata?.user_id || session.client_reference_id || "") as string;
    if (userId) {
      qUpdateUserPaid.run(1, userId);
      qUpdateStripeSessionStatus.run("paid", session.id);
      logEvent(userId, "stripe_webhook_completed", { sessionId: session.id });
    }
  }

  res.json({ received: true });
});
